Here's an implementation of a Bank Account system using JavaScript classes. Each account is created as an instance of the BankAccount class and stored in an array.

Features:

A base class BankAccount for creating accounts.

Stores multiple accounts in an array.

Supports deposit and withdrawal operations.

Displays account details.

---
Implementation:

class BankAccount {
    constructor(accountNumber, accountHolder, balance = 0) {
        this.accountNumber = accountNumber;
        this.accountHolder = accountHolder;
        this.balance = balance;
    }

    deposit(amount) {
        if (amount > 0) {
            this.balance += amount;
            console.log(Deposited ₹${amount} into Account ${this.accountNumber}. New Balance: ₹${this.balance});
        } else {
            console.log("Deposit amount must be greater than zero.");
        }
    }

    withdraw(amount) {
        if (amount > 0 && amount <= this.balance) {
            this.balance -= amount;
            console.log(Withdrawn ₹${amount} from Account ${this.accountNumber}. New Balance: ₹${this.balance});
        } else {
            console.log("Invalid withdrawal amount or insufficient balance.");
        }
    }

    getDetails() {
        return Account No: ${this.accountNumber}, Holder: ${this.accountHolder}, Balance: ₹${this.balance};
    }
}

// Array to store multiple bank accounts
const bankAccounts = [];

// Creating multiple bank accounts
bankAccounts.push(new BankAccount(101, "Aamir", 5000));
bankAccounts.push(new BankAccount(102, "John Doe", 10000));
bankAccounts.push(new BankAccount(103, "Jane Doe", 7500));

// Performing operations
bankAccounts[0].deposit(2000);  // Depositing ₹2000 to Aamir's account
bankAccounts[1].withdraw(3000); // Withdrawing ₹3000 from John Doe's account

// Display all account details
console.log("\nAll Bank Accounts:");
bankAccounts.forEach(account => console.log(account.getDetails()));

---
Expected Output:

Withdrawn ₹3000 from Account 102. New Balance: ₹7000

All Bank Accounts:
Account No: 101, Holder: Aamir, Balance: ₹7000
Account No: 102, Holder: John Doe, Balance: ₹7000
Account No: 103, Holder: Jane Doe, Balance: ₹7500
